cmake_minimum_required(VERSION 3.20)
project(rabbitlands_client_runtime LANGUAGES CXX)
include(FetchContent)
set(GODOT_CPP_REVISION "507ed9d840c01a3c5b2a39af8bb4000bfac30bf5" CACHE STRING "Pinned Godot bindings")
set(GODOTCPP_API_VERSION "4.7" CACHE STRING "" FORCE)
set(GODOTCPP_BUILD_PROFILE "${CMAKE_CURRENT_SOURCE_DIR}/build_profile.json" CACHE FILEPATH "" FORCE)
set(GODOTCPP_SYSTEM_HEADERS ON CACHE BOOL "" FORCE)
set(GODOTCPP_USE_HOT_RELOAD OFF CACHE BOOL "" FORCE)
FetchContent_Declare(godot_cpp GIT_REPOSITORY https://github.com/godotengine/godot-cpp.git GIT_TAG ${GODOT_CPP_REVISION} GIT_SHALLOW FALSE SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/.deps/godot-cpp")
FetchContent_MakeAvailable(godot_cpp)
set(RL_BUILD_JOBS 2 CACHE STRING "Bound native compiler processes")
if(MSVC)
    get_target_property(RL_BINDING_TARGET godot-cpp ALIASED_TARGET)
    if(NOT RL_BINDING_TARGET)
        set(RL_BINDING_TARGET godot-cpp)
    endif()
    get_target_property(RL_BINDING_OPTIONS ${RL_BINDING_TARGET} COMPILE_OPTIONS)
    if(RL_BINDING_OPTIONS)
        string(REGEX REPLACE "/MP[0-9]+" "/MP${RL_BUILD_JOBS}" RL_BINDING_OPTIONS "${RL_BINDING_OPTIONS}")
        set_property(TARGET ${RL_BINDING_TARGET} PROPERTY COMPILE_OPTIONS "${RL_BINDING_OPTIONS}")
    endif()
endif()
if(NOT EXISTS "${RL_RUST_LIBRARY}")
    message(FATAL_ERROR "Build the pinned Rust static library first; pass RL_RUST_LIBRARY")
endif()
add_library(client_runtime SHARED src/client_runtime.cpp src/client_runtime.h src/client_core.h src/mounted_upper_sampler.cpp src/mounted_upper_sampler.h src/mounted_rein_renderer.cpp src/mounted_rein_renderer.h)
target_link_libraries(client_runtime PRIVATE godot-cpp "${RL_RUST_LIBRARY}")
if(WIN32)
    target_link_libraries(client_runtime PRIVATE ws2_32 userenv advapi32 bcrypt ntdll)
endif()
target_compile_features(client_runtime PRIVATE cxx_std_17)
set_target_properties(client_runtime PROPERTIES PREFIX "" OUTPUT_NAME "client_runtime")
