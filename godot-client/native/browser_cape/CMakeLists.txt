cmake_minimum_required(VERSION 3.20)
project(rabbitlands_browser_cape LANGUAGES CXX)
if(NOT EXISTS "${RL_GODOT_CPP}/CMakeLists.txt")
    message(FATAL_ERROR "Pass the pinned Godot bindings directory with RL_GODOT_CPP")
endif()
set(GODOTCPP_API_VERSION "4.7" CACHE STRING "" FORCE)
set(GODOTCPP_BUILD_PROFILE "${CMAKE_CURRENT_SOURCE_DIR}/build_profile.json" CACHE FILEPATH "" FORCE)
set(GODOTCPP_USE_HOT_RELOAD OFF CACHE BOOL "" FORCE)
set(GODOTCPP_SYSTEM_HEADERS ON CACHE BOOL "" FORCE)
add_subdirectory("${RL_GODOT_CPP}" godot-cpp)
add_library(browser_cape SHARED register_types.cpp
    ../client_runtime/src/mounted_cape_body_surface.cpp
    ../client_runtime/src/mounted_cape_projector.cpp)
target_link_libraries(browser_cape PRIVATE godot-cpp)
target_compile_features(browser_cape PRIVATE cxx_std_17)
if(EMSCRIPTEN)
    set(RL_BROWSER_CAPE_WORKERS 4 CACHE STRING "Workers from the browser package's reserved pthread pool")
    if(RL_BROWSER_CAPE_WORKERS LESS 0 OR RL_BROWSER_CAPE_WORKERS GREATER 4)
        message(FATAL_ERROR "Browser cape workers must be between zero and four")
    endif()
    target_compile_definitions(browser_cape PRIVATE RL_BROWSER_CAPE_WORKERS=${RL_BROWSER_CAPE_WORKERS})
    target_link_options(browser_cape PRIVATE -sSIDE_MODULE=1 -pthread -g2 --emit-symbol-map)
    set_target_properties(browser_cape PROPERTIES SUFFIX ".wasm")
endif()
set_target_properties(browser_cape PROPERTIES PREFIX "" OUTPUT_NAME "browser_cape")
